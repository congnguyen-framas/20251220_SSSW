    static Stage New(bool login = true)
    {
        var s = new Stage();
        if (login) s.Login(); else s.RfidConnectedOnly();
        return s;

    static Stage NewSandal(out List<FT600> rows, bool login = true)
    {
        var s = New(login);

    static FT600 R(List<FT600> rows, string name) => rows.First(r => r.C003 == name);

    static void Pick(Stage s, FT600 row)
    {
        s.ClickScale(row);
        if (Hist.TryGetValue(row.C003!, out var h) && !row.IsPu) s.SeedHistory(row, Data.History(h.T, h.P, h.Part, h.Run));
    }

    // ── Chụp + chú thích ───────────────────────────────────────────────────
    static void Shot(Stage s, string name, IEnumerable<Ann>? anns = null, Rect? crop = null, double scale = 1.0, bool dialog = false)
    {
        var img = s.Capture(scale);
        (string, string, string)? d = null;
        if (dialog && s.Dialog.Log.Count > 0) { var l = s.Dialog.Log.Last(); d = (l.Kind, l.Title, l.Message); }
        var res = Annotator.Draw(img, anns ?? Array.Empty<Ann>(), scale, crop, d);
        Annotator.Save(res, Path.Combine(_out, name + ".png"));
        Console.WriteLine("shot " + name);
    }

    static Rect Row(Stage s, FT600 row)
    {
        Stage.Settle();
        var root = (FrameworkElement)s.Win.Content;
        var r = Annotator.Descendants<DataGridRow>(root).First(x => ReferenceEquals(x.Item, row));
        var b = Annotator.BoundsOf(r, root);
        return new Rect(b.X, b.Y, Math.Min(b.Width, 1892 - b.X), b.Height);
    }

    static Rect Cell(Stage s, FT600 row, int dx, int w, int dy = 0, int h = 0)
    {
        var rb = Row(s, row); return new Rect(rb.X + dx, rb.Y + dy, w, h == 0 ? Math.Min(rb.Height, 56) : h);
    }

    static Rect ActionBtn(Stage s, FT600 row, string text)
    {
        Stage.Settle();
        var root = (FrameworkElement)s.Win.Content;
        var r = Annotator.Descendants<DataGridRow>(root).First(x => ReferenceEquals(x.Item, row));
        var b = Annotator.Descendants<Button>(r).First(x => x.Content?.ToString() == text);
        return Annotator.BoundsOf(b, root);
    }

    static Ann A(Rect r, string n) => new(r, n);

    // ════════════════════════════════════════════════════════════════════
    public static void RunAll(string outDir, string? only)
    {
        _out = outDir;
        var all = new (string Name, Action Run)[]
        {
            ("overview", Overview), ("login", Login), ("load-steps", LoadSteps), ("grid-columns", GridColumns),
            ("standard", Standard), ("rex", Rex), ("nonwoven", Nonwoven), ("level2", Level2), ("chain", Chain),
            ("partition", Partition), ("multisize", MultiSize), ("pu", Pu), ("tolerance", Tolerance),
            ("warnings", Warnings), ("confirm", Confirm), ("rowactions", RowActions),
        };
        foreach (var (n, run) in all)
        {
            if (only != null && only != n) continue;
            try { run(); }
            catch (Exception ex) { Console.WriteLine($"!! {n}: {ex}"); }
        }
    }
